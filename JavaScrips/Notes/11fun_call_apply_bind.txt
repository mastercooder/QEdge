function ==> call ,apply , bind
==============================

call method of function
=======================
    call method is used to invoke a function with specified 'this' object vlaue
    we can pass arguments individually
    this is nothing but setting up the context of the function explicitly


function f1(){
    return 'welcome';
}

f1();

function f1(){
    return this.a + this.b ;
}

let obj1 = {a:111,b:222};
let obj2 = {a:300,b:400};

f1.call(obj1);   // 300
f1.call(obj2);  // 700


function f1(c=0,d=0){
    return this.a + this.b + c + d;
}

let obj1 = {a:100,b:200};
let obj2 = {a:300,b:400};
let obj3 = {1:300,2:400};

f1.call(obj1);   //  300
f1.call(obj1,200,300);   //  800
f1.call(obj2,200,300);   //  1200
let ar1 = [11,22];
f1.apply(obj1,ar1);   // 333
f1.call(obj1,...ar1);   // 333

for(cid in customers){

apply
===========================================================
apply method is similar to call method 
apply also used to invoke function with a specified 'this' value as arguments
Only differece between call and apply for passing arguments


var a=10
var b=20
function f1(){
    return this.a + this.b ;
}

f1.apply(this)   //30
f1.apply(window)  //30
f1.apply()  //30

var obj1 = {a:1000,b:2000}

f1.apply(obj1)  //3000

function f1(c=0,d=0){
    return this.a+this.b+c+d
}

let ar1 = [11,22];

f1.apply(this,[...ar1])   
f1.apply(this,[50,60])
