Encapsulation
==============
data hiding ==>  public , protected , private
=============================================
Public : It is accessible from anywhere 
    variableName = 100;
protected: 
    _variable = 100
Private : accessible only from inside the class 
    #variableName = 200

class x{
    a = 0
    set a(v){
        if(v < 0 ) {
            a = 0;
        } else {
            this.a = v;
        }
    }
    get a(){
        return this.a;
    }
    constructor(b){
        this.b = b;
    }
}

let obj1 = new x();
let obj1 = new x(100);
obj1.a = -30;
console.log(obj1);
